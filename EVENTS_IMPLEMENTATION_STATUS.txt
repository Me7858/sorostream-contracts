                       INDEXER BUILD WORKFLOW
================================================================================

Step 1: Subscribe to Events
  import { SorobanRpc } from '@stellar/js-stellar-sdk';
  const server = new SorobanRpc.Server(rpc_url);
  const events = server.subscribeContractEvents({
    contractId: contract_id
  });

Step 2: Parse Event Data
  events.on('message', (event) => {
    const type = event.topics[0];  // Event type
    const streamId = event.topics[1];  // Stream ID
    const data = event.data;  // Event data fields
  });

Step 3: Update Stream State
  switch(type) {
    case 'StreamCreated':
      streams[streamId] = {
        sender: data[0],
        recipient: data[1],
        amount: BigInt(data[2]),
        // ... all fields
      };
      break;
    // ... handle other events
  }

Step 4: Persist to Database
  db.insert('stream_events', {
    stream_id: streamId,
    event_type: type,
    data: JSON.stringify(data),
    ledger_seq: event.ledger.sequence
  });

Step 5: Query from Index
  const stream = await db.query(
    'SELECT * FROM streams WHERE id = ?',
    [streamId]
  );

================================================================================
                          SAMPLE DATABASE QUERIES
================================================================================

Get current stream state:
  SELECT * FROM streams WHERE id = 123;

Calculate current balance:
  SELECT (deposit - total_withdrawn) as balance
  FROM streams WHERE id = 123;

Calculate claimable amount:
  SELECT (flow_rate * (NOW() - start_time)) - total_withdrawn as claimable
  FROM streams WHERE id = 123 AND status = 'active';

Get stream history:
  SELECT event_type, data, created_at
  FROM stream_events
  WHERE stream_id = 123
  ORDER BY ledger_seq;

Find all streams for recipient:
  SELECT * FROM streams
  WHERE recipient = 'GBBBB'
  ORDER BY created_at DESC;

Calculate total fees paid:
  SELECT SUM(CAST(data->>'2' AS BIGINT)) as total_fees
  FROM stream_events
  WHERE stream_id = 123 AND event_type = 'FeeCollected';

Get streams by status:
  SELECT id, sender, recipient, status, deposit, total_withdrawn
  FROM streams
  WHERE status = 'active'
  ORDER BY created_at DESC;
