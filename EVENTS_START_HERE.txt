================================================================================
                      MINIMAL DATABASE SCHEMA
================================================================================

CREATE TABLE streams (
  id BIGINT PRIMARY KEY,
  sender TEXT,
  recipient TEXT,
  amount BIGINT,
  flow_rate BIGINT,
  start_time BIGINT,
  end_time BIGINT,
  status TEXT,           -- 'active', 'paused', 'cancelled', 'completed'
  total_withdrawn BIGINT,
  created_at TIMESTAMP
);

CREATE TABLE stream_events (
  id SERIAL PRIMARY KEY,
  stream_id BIGINT,
  event_type TEXT,
  data JSONB,
  ledger_seq BIGINT UNIQUE,
  created_at TIMESTAMP
);

CREATE INDEX idx_streams_status ON streams(status);
CREATE INDEX idx_streams_recipient ON streams(recipient);
CREATE INDEX idx_events_stream ON stream_events(stream_id);
CREATE INDEX idx_events_ledger ON stream_events(ledger_seq);

                         SAMPLE QUERIES
================================================================================

Get current stream state:
  SELECT * FROM streams WHERE id = 123;

Calculate claimable amount:
  SELECT (flow_rate * (NOW() - start_time)) - total_withdrawn
  FROM streams WHERE id = 123 AND status = 'active';

Get stream history:
  SELECT event_type, data, created_at
  FROM stream_events
  WHERE stream_id = 123
  ORDER BY ledger_seq;

Find active streams for recipient:
  SELECT * FROM streams
  WHERE recipient = 'GBBBB' AND status = 'active';

Calculate total fees for a stream:
  SELECT SUM(data->>'amount'::INT)
  FROM stream_events
  WHERE stream_id = 123 AND event_type = 'FeeCollected';

================================================================================
                          COMMON PATTERNS
================================================================================

Pattern 1: Reconstruct Stream State
  events.forEach(event => {
    if (event.type === 'StreamCreated') {
      stream.deposit = event.amount;
      stream.endTime = event.endTime;
    } else if (event.type === 'StreamWithdrawn') {
      stream.totalWithdrawn += event.amount;
    } else if (event.type === 'StreamCancelled') {
      stream.status = 'cancelled';
    }
  });

Pattern 2: Calculate Current Balance
  const now = Date.now() / 1000;
  const elapsed = now - stream.startTime;
  const accrued = stream.flowRate * elapsed;
  const claimable = accrued - stream.totalWithdrawn;

Pattern 3: Handle Pause/Resume
  const pausedDuration = resumeTime - pauseTime;
  stream.endTime += pausedDuration;
  // Time advancement resumes from new endTime

Pattern 4: Track Event Ordering
  events.sort((a, b) => a.ledger_seq - b.ledger_seq);
  // Process in ledger sequence order
